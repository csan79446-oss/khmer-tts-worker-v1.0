voxcpm @ git+https://github.com/OpenBMB/VoxCPM.git@2.0.3

# VoxCPM declares `torchcodec` (unpinned) but NEVER imports it: a full grep of
# the 2.0.3 sdist for "torchcodec" matches only pyproject.toml / PKG-INFO /
# uv.lock — zero .py files. Left unpinned, pip resolves torchcodec 0.16, which
# is built for torch >= 2.11 and cannot load against the stable torch 2.7.1
# build that ships inside pytorch/pytorch:2.7.1-cuda12.8-cudnn9-runtime. Pin
# the last release built for torch 2.7 (torchcodec 0.5, per PyTorch's
# compatibility table) so the wheel always matches the interpreter's torch ABI.
